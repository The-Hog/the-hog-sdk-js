# PostSearchDto

## Example Usage

```typescript
import { PostSearchDto } from "@the-hog/sdk/models";

let value: PostSearchDto = {
  type: "web_search",
  query: "AI startup funding 2026",
  matchAny: [
    "AI startup",
    "machine learning funding",
  ],
  matchAll: [
    "Series A",
    "2026",
  ],
  exclude: [
    "crypto",
    "blockchain",
  ],
  site: "techcrunch.com",
  hashtag: "productanalytics",
  subreddit: "startups",
};
```
